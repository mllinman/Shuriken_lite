cmake_minimum_required(VERSION 3.16)
project(Shuriken_lite VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Console version
add_executable(Shuriken_lite_console
    main_console.cpp
    src/SimpleBuilder.cpp
    src/SimpleBuilder.h
)

# Web server version
add_executable(Shuriken_lite_web
    main_webserver.cpp
    src/SimpleBuilder.cpp
    src/SimpleBuilder.h
)

# Link threading for web server
find_package(Threads REQUIRED)
target_link_libraries(Shuriken_lite_web Threads::Threads)

# Set output properties
set_target_properties(Shuriken_lite_console PROPERTIES OUTPUT_NAME "shuriken_console")
set_target_properties(Shuriken_lite_web PROPERTIES OUTPUT_NAME "shuriken_web")
